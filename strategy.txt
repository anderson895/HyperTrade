The 15-Minute Volume Rejection Strategy captures short-term market reversals by identifying moments where price aggressively breaks past recent boundaries on high volume, but gets forcefully rejected by liquidity providers.

Core Mechanics & Setup Rules
1. Finding the Playing Field (24-Hour Range)

The bot tracks the highest high and lowest low of the last 96 candles (the past 24 hours of 15-minute price action) to establish clean resistance and support boundaries.

2. Spotting the Fakeout (Volume Spike + Large Wick)

The bot monitors every 15-minute candle close. It looks for a sudden volume spike—specifically, volume that is 2x higher than the recent 20-period moving average.

It checks if the candle pushed beyond the 24-hour High or Low, but left a long wick (at least 40% of the candle's total length), proving that buyers or sellers were heavily rejected.

3. Execution (Post-Only Limit Orders)

Short Signal: Price breaks above the 24-hour high, leaves a long upper wick, and closes back inside the range. The bot places a Post-Only (ALO) limit order at the candle close.

Long Signal: Price breaks below the 24-hour low, leaves a long lower wick, and closes back inside the range. The bot places a Post-Only limit order at the candle close.

Expiration: If the market doesn't pull back to fill the limit order within 30 minutes (2 candles), the order is automatically canceled to avoid chasing stale moves.

Risk Management & Protection
Position Sizing: Risk is dynamically set to 3% of total account equity per trade, with maximum leverage hard-capped at 5x.

Stop-Loss: Placed just beyond the high or low of the rejection wick (roughly 0.38% away from entry).

News Protection Filter: The bot automatically pauses scanning and cancels open limit orders 30 minutes before and 30 minutes after major high-impact USD economic events (such as CPI, NFP, and FOMC) to avoid volatility sweeps.

Isolated Margin: Every trade is strictly isolated, ensuring that even during a complete network or exchange outage, maximum trade loss is hard-capped without threatening your remaining wallet balance.

NEWS_BLACKOUT_AFTER_MINS = 30             # Pause trading 30 mins AFTER high-impact event
TARGET_CURRENCIES = ["USD"]               # Target news currencies
MIN_IMPACT_LEVEL = "High"                 # Filter High impact events only


# =====================================================================
# NEWS CALENDAR FILTER CLASS
# =====================================================================
class NewsFilter:
    """Fetches economic calendar events and checks for active news blackout windows."""

    def __init__(self, before_mins=30, after_mins=30, currencies=None, impact_level="High"):
        self.before_delta = timedelta(minutes=before_mins)
        self.after_delta = timedelta(minutes=after_mins)
        self.currencies = currencies or ["USD"]
        self.impact_level = impact_level
        self.events = []
        self.last_fetch_time = None
        self.fetch_calendar()

    def fetch_calendar(self):
        """Fetch economic events feed (Forex Factory JSON feed)."""
        try:
            url = "https://nodedata.forexfactory.com/forex.json"
            headers = {"User-Agent": "Mozilla/5.0"}
            res = requests.get(url, headers=headers, timeout=10)
            res.raise_for_status()
            data = res.json()

            parsed_events = []
            for item in data:
                if item.get("impact") == self.impact_level and item.get("country") in self.currencies:
                    event_dt = datetime.fromisoformat(item["date"])
                    parsed_events.append({
                        "title": item.get("title"),
                        "country": item.get("country"),
                        "impact": item.get("impact"),
                        "datetime_utc": event_dt.astimezone(timezone.utc)
                    })

            self.events = parsed_events
            self.last_fetch_time = datetime.now(timezone.utc)
            logging.info(f"News Filter: Loaded {len(self.events)} High-Impact news events.")
        except Exception as e:
            logging.error(f"Failed to fetch economic news calendar: {e}")

    def is_news_blackout(self) -> tuple[bool, str]:
        now_utc = datetime.now(timezone.utc)

        if not self.last_fetch_time or (now_utc - self.last_fetch_time).total_seconds() > 43200:
            self.fetch_calendar()

        for event in self.events:
            event_time = event["datetime_utc"]
            window_start = event_time - self.before_delta
            window_end = event_time + self.after_delta

            if window_start <= now_utc <= window_end:
                desc = f"[{event['impact']} Impact] {event['country']} - {event['title']} at {event_time.strftime('%H:%M UTC')}"
                return True, desc

        return False, ""

# HYPERLIQUID BOT ENGINE (WITH TP, SL & TRAILING STOP)

class VolumeRejectionBot:
    def __init__(self, secret_key: str, api_url: str):
        self.account = Account.from_key(secret_key)
        self.address = self.account.address
        self.info = Info(api_url, skip_ws=True)
        self.exchange = Exchange(self.account, api_url)
        
        # Local tracking state for trailing stop activation
        self.highest_price_seen = 0.0
        self.lowest_price_seen = float('inf')
        self.active_sl_price = None

        logging.info(f"Bot Initialized. Wallet: {self.address}")
        self.configure_margin_mode()

        if ENABLE_NEWS_FILTER:
            self.news_filter = NewsFilter(
                before_mins=NEWS_BLACKOUT_BEFORE_MINS,
                after_mins=NEWS_BLACKOUT_AFTER_MINS,
                currencies=TARGET_CURRENCIES,
                impact_level=MIN_IMPACT_LEVEL
            )
        else:
            self.news_filter = None

    def configure_margin_mode(self):
        """Enforces 5x Isolated Margin on Hyperliquid."""
        try:
            is_cross = not ISOLATED_MARGIN
            mode_str = "CROSS" if is_cross else "ISOLATED"
            res = self.exchange.update_leverage(leverage=MAX_LEVERAGE, coin=COIN, is_cross=is_cross)
            if res.get("status") == "ok":
                logging.info(f"Configured {COIN} -> {MAX_LEVERAGE}x {mode_str} Margin.")
        except Exception as e:
            logging.warning(f"Margin mode already configured or notice: {e}")

    def fetch_klines(self, limit: int = 100):
        """Fetch historical 15m OHLCV candles."""
        try:
            url = f"{API_URL}/info"
            payload = {
                "type": "candleSnapshot",
                "req": {"coin": COIN, "interval": TIMEFRAME, "startTime": int((time.time() - limit * 900) * 1000)}
            }
            res = requests.post(url, json=payload, timeout=10)
            res.raise_for_status()
            candles = res.json()
            return [{"open": float(c['o']), "high": float(c['h']), "low": float(c['l']), "close": float(c['c']), "volume": float(c['v'])} for c in candles]
        except Exception as e:
            logging.error(f"Error fetching candles: {e}")
            return []

    def get_current_price(self) -> float:
        """Fetches the latest mid/market price for BTC."""
        try:
            mids = self.info.all_mids()
            return float(mids[COIN])
        except Exception as e:
            logging.error(f"Error fetching current price: {e}")
            return 0.0

    def get_open_position(self):
        """Checks Hyperliquid for active open positions on BTC."""
        try:
            user_state = self.info.user_state(self.address)
            for pos in user_state.get("assetPositions", []):
                position_data = pos["position"]
                if position_data["coin"] == COIN:
                    szi = float(position_data["szi"])
                    if szi != 0:
                        return {
                            "size": szi,
                            "entry_price": float(position_data["entryPx"]),
                            "side": "LONG" if szi > 0 else "SHORT",
                            "unrealized_pnl": float(position_data["unrealizedPnl"])
                        }
            return None
        except Exception as e:
            logging.error(f"Error checking position state: {e}")
            return None

    def analyze_volume_rejection(self, candles):
        """Calculates entry, initial SL, and fixed Take-Profit levels."""
        if len(candles) < RANGE_LOOKBACK + 1:
            return None

        target_candle = candles[-2]
        history = candles[-(RANGE_LOOKBACK + 2):-2]

        vol_ma = sum(c['volume'] for c in candles[-22:-2]) / 20.0
        if target_candle['volume'] < (vol_ma * VOL_MULT):
            return None

        range_high = max(c['high'] for c in history)
        range_low = min(c['low'] for c in history)
        candle_range = target_candle['high'] - target_candle['low']
        if candle_range == 0:
            return None

        upper_wick = target_candle['high'] - max(target_candle['open'], target_candle['close'])
        lower_wick = min(target_candle['open'], target_candle['close']) - target_candle['low']

        # Bearish Rejection (SHORT)
        if target_candle['high'] >= range_high and (upper_wick / candle_range) >= 0.4 and target_candle['close'] < range_high:
            entry = round(target_candle['close'], 1)
            sl = round(target_candle['high'] * 1.001, 1)
            risk_dist = abs(entry - sl)
            tp = round(entry - (risk_dist * FIXED_TP_RR), 1)
            return {"side": "SHORT", "entry_limit": entry, "stop_loss": sl, "take_profit": tp}

        # Bullish Rejection (LONG)
        if target_candle['low'] <= range_low and (lower_wick / candle_range) >= 0.4 and target_candle['close'] > range_low:
            entry = round(target_candle['close'], 1)
            sl = round(target_candle['low'] * 0.999, 1)
            risk_dist = abs(entry - sl)
            tp = round(entry + (risk_dist * FIXED_TP_RR), 1)
            return {"side": "LONG", "entry_limit": entry, "stop_loss": sl, "take_profit": tp}

        return None

    def get_account_equity(self) -> float:
        try:
            user_state = self.info.user_state(self.address)
            return float(user_state["marginSummary"]["accountValue"])
        except Exception as e:
            logging.error(f"Error fetching account equity: {e}")
            return 0.0

    def calculate_position_size(self, entry_price: float, stop_loss: float, equity: float) -> float:
        sl_dist_pct = abs(entry_price - stop_loss) / entry_price
        if sl_dist_pct == 0:
            return 0.0

        risk_amount = equity * ACCOUNT_RISK_PCT
        position_notional = min(risk_amount / sl_dist_pct, equity * MAX_LEVERAGE)
        return round(position_notional / entry_price, 4)

    def place_bracket_orders(self, side: str, sz: float, stop_loss: float, take_profit: float):
        """Places native Trigger Stop-Loss and Take-Profit orders on Hyperliquid."""
        is_buy_exit = (side == "SHORT")  # To exit a SHORT, we BUY. To exit a LONG, we SELL.

        # 1. Native Stop-Loss Order
        sl_order_type = {"trigger": {"isMarket": True, "triggerPx": stop_loss, "tpsl": "sl"}}
        self.exchange.order(coin=COIN, is_buy=is_buy_exit, sz=sz, limit_px=stop_loss, order_type=sl_order_type, reduce_only=True)
        self.active_sl_price = stop_loss
        logging.info(f"Native Stop-Loss Order Set @ ${stop_loss}")

        # 2. Native Take-Profit Order
        tp_order_type = {"trigger": {"isMarket": True, "triggerPx": take_profit, "tpsl": "tp"}}
        self.exchange.order(coin=COIN, is_buy=is_buy_exit, sz=sz, limit_px=take_profit, order_type=tp_order_type, reduce_only=True)
        logging.info(f"Native Take-Profit Order Set @ ${take_profit}")

    def update_trigger_stop(self, is_buy: bool, sz: float, new_sl_price: float):
        """Updates the resting trigger stop order to trail price."""
        sl_order_type = {"trigger": {"isMarket": True, "triggerPx": new_sl_price, "tpsl": "sl"}}
        res = self.exchange.order(coin=COIN, is_buy=is_buy, sz=sz, limit_px=new_sl_price, order_type=sl_order_type, reduce_only=True)
        if res.get("status") == "ok":
            self.active_sl_price = new_sl_price
            logging.info(f"TRAILING STOP UPDATED: New Stop-Loss @ ${new_sl_price}")

    def manage_active_position(self, position):
        """Monitors active position and updates the trailing stop as profits grow."""
        current_price = self.get_current_price()
        if current_price == 0.0:
            return

        side = position["side"]
        entry = position["entry_price"]
        sz = abs(position["size"])
        is_buy_exit = (side == "SHORT")

        # Track price extremes during trade lifecycle
        if side == "LONG":
            self.highest_price_seen = max(self.highest_price_seen, current_price)
            profit_dist = self.highest_price_seen - entry
            
            # Check if trailing activation threshold is met
            if ENABLE_TRAILING_STOP and profit_dist > 0:
                trail_sl = round(self.highest_price_seen * (1 - TRAILING_DISTANCE_PCT), 1)
                
                # Only raise stop-loss if it locks in more profit than current active SL
                if self.active_sl_price is None or trail_sl > self.active_sl_price:
                    if trail_sl > entry:  # Ensure we are in profit
                        self.update_trigger_stop(is_buy=is_buy_exit, sz=sz, new_sl_price=trail_sl)

        elif side == "SHORT":
            self.lowest_price_seen = min(self.lowest_price_seen, current_price)
            profit_dist = entry - self.lowest_price_seen
            
            # Check if trailing activation threshold is met
            if ENABLE_TRAILING_STOP and profit_dist > 0:
                trail_sl = round(self.lowest_price_seen * (1 + TRAILING_DISTANCE_PCT), 1)
                
                # Only lower stop-loss if it locks in more profit than current active SL
                if self.active_sl_price is None or trail_sl < self.active_sl_price:
                    if trail_sl < entry:  # Ensure we are in profit
                        self.update_trigger_stop(is_buy=is_buy_exit, sz=sz, new_sl_price=trail_sl)

    def cancel_all_open_orders(self):
        try:
            open_orders = self.info.open_orders(self.address)
            for o in open_orders:
                if o["coin"] == COIN:
                    self.exchange.cancel(COIN, o["oid"])
                    logging.info(f"Canceled resting order {o['oid']}")
        except Exception as e:
            logging.error(f"Error canceling open orders: {e}")

    def execute_post_only_limit_order(self, setup, btc_size):
        is_buy = (setup['side'] == "LONG")
        order_type = {"limit": {"tif": "Alo"}}

        logging.info(f"Placing Post-Only Limit Order: {setup['side']} {btc_size} BTC @ ${setup['entry_limit']}")

        res = self.exchange.order(coin=COIN, is_buy=is_buy, sz=btc_size, limit_px=setup['entry_limit'], order_type=order_type, reduce_only=False)

        if res.get("status") == "ok":
            status_data = res["response"]["data"]["statuses"][0]
            if "resting" in status_data:
                oid = status_data["resting"]["oid"]
                logging.info(f"Order resting on book. OID: {oid}")
                
                # Attach bracket SL & TP immediately
                self.place_bracket_orders(setup['side'], btc_size, setup['stop_loss'], setup['take_profit'])
                
                # Reset trailing tracking variables
                self.highest_price_seen = setup['entry_limit']
                self.lowest_price_seen = setup['entry_limit']
                return oid
            elif "error" in status_data:
                logging.warning(f"Post-Only rejected by engine: {status_data['error']}")
        else:
            logging.error(f"Order submission failed: {res}")
        return None

    def run(self):
        logging.info("Starting Volume Rejection Bot (Position Tracker + SL/TP + Trailing Stop Engine)...")

        while True:
            try:
                # 1. Active Position Tracking & Trailing Management
                open_pos = self.get_open_position()
                if open_pos:
                    logging.info(f"ACTIVE POSITION: {open_pos['side']} {abs(open_pos['size'])} BTC | Entry: ${open_pos['entry_price']} | uPNL: ${open_pos['unrealized_pnl']:.2f}")
                    self.manage_active_position(open_pos)
                else:
                    # Reset tracker when flat
                    self.highest_price_seen = 0.0
                    self.lowest_price_seen = float('inf')
                    self.active_sl_price = None

                # 2. Check News Blackout Window
                if ENABLE_NEWS_FILTER and self.news_filter:
                    is_blackout, event_info = self.news_filter.is_news_blackout()
                    if is_blackout:
                        logging.warning(f"NEWS BLACKOUT ACTIVE: {event_info}")
                        self.cancel_all_open_orders()
                        time.sleep(60)
                        continue

                # 3. Strategy Evaluation (Only evaluate new entries if no position is active)
                if not open_pos:
                    candles = self.fetch_klines(limit=100)
                    if candles:
                        setup = self.analyze_volume_rejection(candles)
                        if setup:
                            logging.info(f"VALID SETUP DETECTED: {setup['side']} @ ${setup['entry_limit']} | TP: ${setup['take_profit']} | SL: ${setup['stop_loss']}")
                            equity = self.get_account_equity()
                            if equity > 0:
                                btc_size = self.calculate_position_size(setup['entry_limit'], setup['stop_loss'], equity)
                                if btc_size > 0:
                                    self.execute_post_only_limit_order(setup, btc_size)

                # Wait for next check boundary
                time.sleep(15)

            except Exception as e:
                logging.error(f"Error in main loop: {e}")
                time.sleep(15)
